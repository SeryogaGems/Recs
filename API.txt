basic-component = {
    [1] = buffer.create(64+, 0x64) - quad-chunk-bitmap (uint_64)
    [2] = 0 | { function, function, function },
    [3] = { N-chunk [dense]: buffer.create([0x16], 0x40) - x16 [block 0 and block fields 1..4] (80 uint_64) },
    [4] = { N-chunk x 16 [optional dense]: buffer.create([id], 0x256) - [entity fields 0..255] (256 uint_32) },
    [5] = { N-chunk x 16 [optional dense]: table.create([256]) - [value fields 1..256] (256 TValue) }
} => 84.784 (4096) / ~82Kb or 1952 bytes when empty

multi-component = {

    [6?] = { N-chunk x 16 [optional dense]: table.create([256]) - [value fields 1..256] (256 TValue) },

//MASK STANDARDS//

Etype - [0..15] - segment_cache => 16 bits (number)

Entity - [0..20] - chunk and segment + [20..21] - block + [21..31] - bitmap + [56..63] - cluster => 64 bits (integer)

Group - [0..3] - hooks + [4..27] - relation + [28..51] - target => 52 bits (integer)

Multi - [4..7] - table count => 8 bits (number?)

Component - [0..3] - life => 12 bits (number?)

Tag - [0..3] - life => 4 bits (number?)

//API//
world(count: number?): Id - создать объект мира. Это точка входа в Recs. =>
    [
        0. Проверяем, не создан ли мир через специальный флаг
        1. Подсчитываем количество сущностей в проекте 
        2. Создаём таблицы: сущностей, компонентов (разряженный и плотный), кэши графа, кэш сегментов и кэш запросов;
        3. Добавляем таблицы WORLD_ARRAY
        4. Замораживаем таблицу и меняем флаг создания
    ]

etype(world: World): Id - создать новый тип объекта. Генерируются вплоть до 245. Далее - ошибка.
Автоматически расширяются и добавляются в cache-часть мира. =>
    [
        1. Вычисляем id нового типа через переменную счётчик. Если больше 256 - выдаём ошибку.
        2. Заносим id типа в таблицу сущностей с маской равной id
        3. Заходим в meta компонент и определяем через # старший чанк
        4. Считываем маску чанка и смотрим старший нулевой бит через not + counrz методы. Если он больше 16 - присваиваем максимальный.
        5. Создаём в метакомпоненте новый буфер значений: маска сегмента x маски блоков x id-типа
        6. Создаём буфер в таблице кэшей сегментов и заносим туда № сегмента по формуле (chunk (начало от 0) x 16 + segment)
        7. Возвращаем id
    ]

migrate(world: World, entity: Entity, type: EType): () - переместить сущность из одного типа в другой.

entity(world: World, type: EType, custom: number?): Id - создать новую сущность определённого типа. Дополнительно можно указать собственный номер id. 
Однако рекомендуется использовать пакетизацию. =>
    [
        1. Вычисляем id новой сущности через итерацию свободных id. Итерация начинается с младшего значения счётчика сущностей. Либо берём ID пользовательский.
        2. Заходим в кэш сегментов и берём последний элемент кэша.
        3. Превращаем № сегмента в структуру chunk, segment через формулу (cache_segment // 16) и (cache_segement - chunk * 16)
        4. Заходим в метакомпонент, в нужный чанк и начинаем спуск: чанк - сегмент - блок. В последнем ищем свободное место. 
        Спуск осуществляется через not + countrz методы. Постепенно дозаполняем маску entity.
        5. Разворачиваем цикл назад. Если система полная - ставим нужные биты. Нет - выходим.
        5.1. Если сегмент полный: создаём новый сегмент или чанк (при полноте сегментов). Записываем новый id-сегмента в кэш по type_id.
    ]

destroy(world: World, entity: Entity): () - уничтожить сущность или же привести её маску к 0x0. Автоматически запустит метод clean().
Тяжёлый метод по скорости.

clean(world: World, entity: Entity): () - очистить все компоненты. Маска сущности останется прежней.
В случае, если сущность была связью - все отношения с ней будут уничтожены. Тяжёлый метод по скорости.

component(world: World): Id - создать новый базовый компонент с 1 чанком. Генерируются вплоть до 256. Далее - ошибка. =>
    [
        1. Вычисляем id нового компонента. Если привысило 256 - отправляем сообщение об ошибке.
        2. Заносим id типа в таблицу сущностей с маской равной 0b1000 + рандомные 8 бит
        3. Создаём структуру в dense-массиве через #
        4. Смотрим максимальный чанк в проекте и массово создаём множество чанков
        5. Делаем ссылку в sparse: id - sparse_index
        6. Возвращаем id
    ]

tag(world: World): Id - создать новый тег с 1 чанком. Генерируются в пределе 2^28 сущностей. => 
    [
        1. Вычисляем id новой сущности через итерацию свободных id. Итерация начинается с младшего значения счётчика сущностей.
        2. Заносим id типа в таблицу сущностей с маской равной 0b1000
        3. Создаём структуру в dense-массиве через #
        4. Смотрим максимальный чанк в проекте и массово создаём множество чанков
        5. Делаем ссылку в sparse: id - sparse_index
        6. Возвращаем id
    ]

multi(world: World, count: number?): Id - создать новый мульти-компонент с 1 чанком. Генерируются в пределе 2^28 сущностей.
В сигнатуре указывается число дополнительных массивов данных. Максимально - 6 штук. При отсутствии генерируется - 2 штуки.
    [
        1. Вычисляем id новой сущности через итерацию свободных id. Итерация начинается с младшего значения счётчика сущностей.
        2. Заносим id типа в таблицу сущностей с маской равной 0b1000 + число таблиц до 4. Если ничего - создаём 2.
        3. Создаём структуру в dense-массиве через #
        4. Смотрим максимальный чанк в проекте и массово создаём множество чанков
        5. Делаем ссылку в sparse: id - sparse_index
        6. Возвращаем id
    ]

add(world: World, entity: Entity, component: Component | MultiComponent | Group | Tag): () - добавить к сущности компонент, мульти-компонент, тег или группу. =>
    [
        1. Взять ID сущности и ID компонента из аргументов. Найти их маски в сущностях.
        2. Получить данные сущности и компонента
        3. Зайти в чанк. Посмотреть сегмент через прямую ссылку с добавлением 1. Если сегмента нет - создать структуру через проверку лимита числа:
        - Классическую для компонентов и групп
        - Расширенную по старшим битам для мульти
        - Буферную для тегов
        5. Заполнить все маски.
    ]

remove(world: World, entity: Entity, component: Component | MultiComponent | Group | Tag): () - удалить у сущности компонент, мульти-компонент, тег или группу. =>
    [
        1. Взять ID сущности и ID компонента из аргументов. Найти их маски в сущностях.
        2. Получить данные сущности и компонента
        3. Зайти в чанк. Спуститься в нужный блок и инвертировать бит.
        4. Зачистить данные в таблицах, если это не тег.
    ]

set(world: World, entity: Entity | Component | MultiComponent | Group, component: Component | MultiComponent | Group | Special, value: any | () -> (), index: number?): () -
добавить значение для сущности по определённому компоненту, мульти-компоненту, группе. Не доступно для тегов.
Если первый параметр указан как компонент, мульти-компонент или группа - присваивается один из специальных ID хуков. Аргументов value станет callback функция.
Index указывает в какую таблицу записать информацию. По умолчанию 1.

get(world: World, entity: Entity, component: Component | MultiComponent | Group, index: number?): any - получить значение сущности по компоненту,
мульти-компоненту, группе. Поддерживает index для считывания конкретной таблицы. По умолчанию 1. =>
    [
        1. Взять ID сущности и ID компонента из аргументов. Найти их маски в сущностях.
        2. Получить данные сущности и компонента
        3. Зайти в чанк. Спуститься в нужный блок и проверить offset в буфере (??)
        4. Если offset не 0 - вернуть значение в таблице по offset (если указан index, то таблицей будет N index'а)
    ]

group(world: World, relationship: Entity<>, target: Entity<>, ...Entity<Target>): Group<A, B> - создать группу из двух сущностей. Сущностями могут быть сущности, 
компонентные типы, специальные объекты. Группа не поддерживает пока другую группу. Генерирует ID = маске компонента X пересечение двух типов. 
Укладывается в cache-часть мира по пересечению в качестве ключа. При использовании Recs.Wildcard вернёт первую найденую группу. 
Поддерживает в будущем добавление смежных-целей равного типа основной.

target(world: World, source: Entity, relationship: Entity<>, index: number?): Entity<> - получить цель группы по связи. Метод найдёт ключ-группы и пройдётся по его буферам.
В случае успеха вернёт цель. Index отмечает какую из групп вернуть, если отношение состоит из ветвления Recs.ChildOf. Тяжёлый метод по скорости.
    [
        1. Зайти в кэш-графа таргетов. Найти ID буфер. Если нет - вернуть nil
        2. Начать перебирать буфер. Каждый ID проверить через связку: sparse -> dense.
        3. Если найдено совпадение - окончить цикл. Или продолжить при отличном index.
        4. Посмотреть маску ID в сущностях
        5. Вернуть 2 часть группы.ы
    ]

has(world: World, entity: Entity, link: Tag | Group): boolean - проверить есть ли у сущности этот тег или группа. =>
    [
        1. Взять ID сущности и ID компонента из аргументов. Найти их маски в сущностях.
        2. Проверить через сравнение числа вид компонентной сущности.
        3. Спуститься вниз по компоненту. Если тег - буферы, если группа - таблицы.
        4. Вернуть true, если место сущности найдено. False при любом несоответствии.
    ]

query(world: World, ...Component | Group): Query - собрать запрос по компонентам или группе. 
Запрос собирается по квадрантам (4 чанка сразу за 1 чтение флага)

forEach(query: Query): () -> () - итерация запроса. Вернёт связку entity x v1...

with(world: World, query: Query, ...Component | Tag | MultiComponent | Group): () -> () - итерация запроса с добавлением определённого компонента, тега, 
мульти-компонента или группы. Вернёт связку entity x v1...Не меняет значения v. Только эвристит маски.

without(world: World, query: Query, ...Component | Tag | MultiComponent | Group): () -> () - итерация запроса без определённого компонента, тега, 
мульти-компонента или группы. Вернёт связку entity x v1...Не меняет значения v. Только эвристит маски.

iter(world: World, query: Query, ...Component | Tag | MultiComponent | Group): () -> () - итерация запроса. Вернёт только сущности.

-- ? all(world: World, pattern: RecsSpecial | EType): () -> () - вернуть сложные объекты специального типа: компонент объекты (Recs.Component),
типы (Recs.Type), все entity (Recs.All)

cacheQuery(world: World, query: Query): Id<Query> - закэшировать запрос для переиспользования без сборки. Кладётся в cache-часть мира.

-- ? last(world: World, multi: MultiComponent): any - получить последний массив данных у мульти-компонента

-- ? first(world: World, multi: MultiComponent): any - получить первый массив данных у мульти-компонента

find(world: World, key: string): Id? - найти Id-объекта по заданному ключу во внутреннее хранилище.

packed(world: World, key: string, object: Entity<>?): () - сохранить Id-объекта по специальному ключу. 
В случае удаления вместо значения вписать nil во внутреннее хранилище.
